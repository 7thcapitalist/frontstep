import { cp, copyFile, readFile, writeFile } from 'node:fs/promises';
import { createHash } from 'node:crypto';
import path from 'node:path';

export const motionSkills = ['gsap-core', 'gsap-timeline', 'gsap-scrolltrigger', 'gsap-performance', 'gsap-plugins', 'emil-design-eng', 'improve-animations', 'threejs-shaders', 'threejs-textures', 'threejs-interaction', 'threejs-postprocessing'];
export const suppliedSkills = ['frontend-design', 'react-best-practices', 'web-design-guidelines', 'taste-skill', 'soft-skill', 'redesign-skill', 'minimalist-skill', 'ui-ux-pro-max', 'composition-patterns', 'gpt-tasteskill'];
export const weights = { artDirection: .20, typography: .15, composition: .15, imagery: .10, motion: .15, interaction: .15, responsive: .10 };

// Copy complete skill folders: relative references (AUDIT.md etc.) must resolve.
export async function installMotionResources(kit, site) {
  const ref = path.join(site, 'reference');
  const manifest = [];
  for (const name of [...suppliedSkills, ...motionSkills]) {
    const source = path.join(kit, 'skills', name);
    const group = motionSkills.includes(name) ? 'motion-skills' : 'skills';
    const target = path.join(ref, group, name);
    await cp(source, target, { recursive: true });
    const bytes = await readFile(path.join(target, 'SKILL.md'));
    manifest.push({ id: name, path: `reference/${group}/${name}/SKILL.md`, sha256: createHash('sha256').update(bytes).digest('hex') });
  }
  await copyFile(path.join(kit, 'ACCEPTANCE-EVIDENCE.md'), path.join(ref, 'ACCEPTANCE-EVIDENCE.md'));
  await copyFile(path.join(kit, 'VISUAL-RUBRIC.md'), path.join(ref, 'VISUAL-RUBRIC.md'));
  await copyFile(path.join(kit, 'VERCEL-BENCHMARKS.md'), path.join(ref, 'VERCEL-BENCHMARKS.md'));
  await copyFile(path.join(kit, 'CREATIVE-DEVELOPMENT.md'), path.join(ref, 'CREATIVE-DEVELOPMENT.md'));
  await copyFile(path.join(kit, 'SPATIAL-STORYTELLING.md'), path.join(ref, 'SPATIAL-STORYTELLING.md'));
  await cp(path.join(kit, 'spatial-source'), path.join(ref, 'spatial-source'), {recursive:true});
  await copyFile(path.join(kit, 'skills', 'THREEJS-SOURCE.json'), path.join(ref, 'motion-skills', 'THREEJS-SOURCE.json'));
  await copyFile(path.join(kit, 'skills', 'THREEJS-UPSTREAM-README.md'), path.join(ref, 'motion-skills', 'THREEJS-UPSTREAM-README.md'));
  await cp(path.join(kit, 'benchmarks'), path.join(ref, 'benchmarks'), { recursive: true });
  await cp(path.join(kit, 'runtime', 'public'), path.join(site, 'public', 'vendor'), { recursive: true });
  await copyFile(path.join(kit, 'runtime', 'MOTION-RUNTIME.md'), path.join(ref, 'MOTION-RUNTIME.md'));
  await writeFile(path.join(ref, 'skill-manifest.json'), JSON.stringify(manifest, null, 2));
  return manifest;
}

export async function verifySkillUsage(site) {
  const usage = JSON.parse(await readFile(path.join(site, 'skill-usage.json'), 'utf8'));
  if (!Array.isArray(usage) || usage.length < 3) throw new Error('skill-usage.json needs at least three concrete applications');
  for (const row of usage) {
    if (!row.skill || String(row.principle || '').length < 12 || String(row.application || '').length < 20 || !Array.isArray(row.files) || !row.files.length) throw new Error('Incomplete skill application evidence');
    for (const file of row.files) {
      if (!/^public\/[\w./-]+\.(html|css|js)$/.test(file) || file.includes('..')) throw new Error('Skill application must reference local public source');
      await readFile(path.join(site, file));
    }
  }
  return usage;
}

export function assessReview(review) {
  const entries = Object.entries(weights);
  if (!review.criteria || entries.some(([key]) => !Number.isFinite(review.criteria[key]) || review.criteria[key] < 0 || review.criteria[key] > 10)) throw new Error('Visual reviewer omitted calibrated criteria');
  // Compare the weighted value before presentation rounding (9.05 must remain eligible).
  const computed = entries.reduce((sum, [key, weight]) => sum + review.criteria[key] * (weight * 100), 0) / 100;
  return { ...review, reviewerScore: review.score, score: computed, displayScore: computed.toFixed(2), passed: review.passed === true && computed > 9 && entries.every(([key]) => review.criteria[key] >= 8.5) };
}

// A 3D plan must describe actual poses and camera decisions before implementation.
export function verifySpatialPlan(plan) {
  const recipe = String(plan.motionRecipe || '');
  // A reference such as "sem carregar Three.js" declares a DOM implementation.
  // Remove only directly negated technology names, not unrelated positive uses.
  const positiveRecipe = recipe.replace(/\b(?:without|no|sem|n[aã]o|do not|not using|avoid)\s+(?:(?:use|load|usar|carregar|using|loading|importing)\s+)?(?:webgl|three\.?js|kinetic-sculpture|frontstep-spatial)\b/gi, '');
  const mentionsSpatial = /webgl|three\.?js|kinetic-sculpture|frontstep-spatial/i.test(positiveRecipe);
  if (plan.renderingMode !== undefined && !['dom', 'webgl'].includes(plan.renderingMode)) throw new Error('renderingMode must be dom or webgl');
  if (plan.renderingMode === 'dom' && (plan.spatialPlan || mentionsSpatial)) throw new Error('DOM plan conflicts with a positive spatial implementation');
  const required = !!plan.spatialPlan || plan.renderingMode === 'webgl' || mentionsSpatial;
  if(!required)return null;
  const spatial=plan.spatialPlan;
  const fields=['businessMetaphor','firstViewport','assets','mobile','fallback','budget','continuation'];
  if(!spatial||fields.some(key=>typeof spatial[key]!=='string'||spatial[key].trim().length<12))throw new Error('Spatial plan needs business, composition, assets, mobile, fallback, budget and whole-page continuation');
  if(!Array.isArray(spatial.states)||spatial.states.length<3||spatial.states.some(state=>['trigger','geometry','camera','purpose'].some(key=>typeof state[key]!=='string'||state[key].trim().length<8)))throw new Error('Spatial plan needs three concrete trigger/geometry/camera/purpose states');
  if(new Set(spatial.states.map(state=>state.geometry.trim().toLowerCase())).size<3)throw new Error('Spatial states must describe three different geometric compositions');
  return spatial;
}
