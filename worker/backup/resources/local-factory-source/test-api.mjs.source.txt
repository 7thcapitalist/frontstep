#!/usr/bin/env node
import assert from 'node:assert/strict';
import { spawn } from 'node:child_process';
import { randomUUID } from 'node:crypto';
import { readFile, mkdir, writeFile } from 'node:fs/promises';
import path from 'node:path';
import { fileURLToPath } from 'node:url';

const here = path.dirname(fileURLToPath(import.meta.url));
const port = 4191;
const base = `http://127.0.0.1:${port}`;
let server;
function start() {
  server = spawn(process.execPath, [path.join(here, 'server.mjs')], {
    cwd: here, env: { ...process.env, FRONTSTEP_LOCAL_PORT: String(port), FRONTSTEP_CODEX: 'missing-codex-executable-for-test' }, windowsHide: true, stdio: ['ignore', 'pipe', 'pipe']
  });
  server.stderr.on('data', data => process.stderr.write(data));
}
async function waitForReady() {
  for (let i = 0; i < 150; i++) {
    try { if ((await fetch(`${base}/health`)).ok) return; } catch { /* starting */ }
    await new Promise(resolve => setTimeout(resolve, 200));
  }
  throw new Error('Server unavailable');
}
async function stop() {
  if (!server) return;
  const child = server;
  const done = new Promise(resolve => child.once('exit', resolve));
  child.kill('SIGTERM');
  await done;
  server = null;
}
async function post(url, data) {
  const response = await fetch(`${base}${url}`, { method: 'POST', headers: { 'content-type': 'application/json' }, body: JSON.stringify(data) });
  return { code: response.status, body: await response.json() };
}
async function waitFor(id, state) {
  for (let i = 0; i < 100; i++) {
    const job = await (await fetch(`${base}/jobs/${id}`)).json();
    if (job.state === state) return job;
    await new Promise(resolve => setTimeout(resolve, 100));
  }
  throw new Error(`Job ${id} did not reach ${state}`);
}
try {
  start(); await waitForReady();
  const invalid = await post('/jobs', { brief: { slug: '../escape', synthetic_fixture: true } });
  assert.equal(invalid.code, 400);
  const brief = JSON.parse(await readFile(path.join(here, '..', 'local-briefs', 'contabilidade-vila-clara.json'), 'utf8'));
  const key = `api-test-${randomUUID()}`;
  const first = await post('/jobs', { brief, idempotencyKey: key });
  assert.equal(first.code, 202);
  const failed = await waitFor(first.body.id, 'failed');
  assert.match(failed.error, /spawn/);
  const duplicate = await post('/jobs', { brief, idempotencyKey: key });
  assert.equal(duplicate.code, 200); assert.equal(duplicate.body.id, first.body.id);
  const conflict = await post('/jobs', { brief: { ...brief, changed: true }, idempotencyKey: key });
  assert.equal(conflict.code, 409);
  for (let n = 0; n < 2; n++) {
    const retry = await post(`/jobs/${first.body.id}/retry`, {});
    assert.equal(retry.code, 202);
    await waitFor(first.body.id, 'failed');
  }
  const exhausted = await post(`/jobs/${first.body.id}/retry`, {});
  assert.equal(exhausted.code, 409);
  await stop();
  const pendingId = randomUUID(), runningId = randomUUID();
  for (const [id, state] of [[pendingId, 'queued'], [runningId, 'running']]) {
    const dir = path.join(here, 'jobs', id);
    await mkdir(dir, { recursive: true });
    await writeFile(path.join(dir, 'job.json'), JSON.stringify({ id, key: `restart-${id}`, requestHash: id, brief, scenario: null, state, createdAt: new Date().toISOString(), attempt: 0, retryCount: 0 }));
  }
  start(); await waitForReady();
  const persisted = await (await fetch(`${base}/jobs/${first.body.id}`)).json();
  assert.equal(persisted.state, 'failed'); assert.equal(persisted.retryCount, 2);
  await waitFor(pendingId, 'failed');
  const interrupted = await (await fetch(`${base}/jobs/${runningId}`)).json();
  assert.equal(interrupted.state, 'interrupted');
  console.log(JSON.stringify({ passed: true, tests: ['invalid slug', 'spawn failure', 'duplicate key', 'key conflict', 'bounded retry', 'restart persistence', 'queued restart', 'running interruption'], jobId: first.body.id }, null, 2));
} catch (error) { console.error(error); process.exitCode = 1; }
finally { await stop(); }
