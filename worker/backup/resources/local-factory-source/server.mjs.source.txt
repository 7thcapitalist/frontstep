  const text = String(niche || '').toLowerCase();

  if (/advog|jur[ií]d|contab|despach|tribut[aá]ri/.test(text)) return 'juridico_contabil';

  const temp = `${file}.${randomUUID()}.tmp`;
  await writeFile(temp, JSON.stringify(value, null, 2) + '\n');

}

async function publicDigest(root) {
  const hash = createHash('sha256');
  async function walk(folder, prefix = '') {
    for (const entry of (await readdir(folder, { withFileTypes: true })).sort((a, b) => a.name.localeCompare(b.name))) {
      const relative = prefix + entry.name;
      if (entry.isDirectory()) await walk(path.join(folder, entry.name), relative + '/');
      else if (entry.isFile()) { hash.update(relative + '\0'); hash.update(await readFile(path.join(folder, entry.name))); }
    }
  }
  await walk(root);
  return hash.digest('hex');
}

async function bodyJson(req) {
  let body = '';
  for await (const chunk of req) {
    body += chunk;
    if (body.length > 100_000) throw new Error('Body exceeds 100 KB');
  }
  return JSON.parse(body);
}
function validate(input) {
  if (!input || typeof input !== 'object' || Array.isArray(input)) throw new Error('JSON object required');
  const { brief, scenario, idempotencyKey } = input;
  if (!brief || typeof brief !== 'object' || Array.isArray(brief)) throw new Error('brief object required');
  if (typeof brief.slug !== 'string' || !/^[a-z][a-z0-9-]{2,48}$/.test(brief.slug)) throw new Error('Invalid brief.slug');
  if (brief.mode !== 'demo' || brief.synthetic_fixture !== true) throw new Error('Only synthetic demo briefs are accepted');
  const niche = String(brief.business?.niche || '').toLowerCase();
  const food = /restaurante|hamburguer|pizzaria|lanchonete|padaria|confeitaria|açaí|acai|bar\b|caf[eé]|marmita|food truck/.test(niche);
  const photos = brief.media?.photos || [];
  if (!Array.isArray(photos) || photos.length > 12) throw new Error('Invalid media.photos: expected up to 12 photos');
  if (food && !photos.length) throw new Error('Missing media.photos: food demos need at least one local image asset before building');
  if (food && !photos.some(photo => ['business_photo', 'licensed_stock'].includes(photo?.provenance))) {
    throw new Error('Food demos need at least one real business or licensed-stock photograph; generated art alone is insufficient');
  }
  for (const photo of photos) {
    if (!photo || typeof photo.local_path !== 'string' || !/^[a-zA-Z0-9_./-]+\.(jpe?g|png|webp)$/i.test(photo.local_path)
      || photo.local_path.startsWith('/') || photo.local_path.includes('..')) {
      throw new Error('Invalid media.photos: use local image files under frontstep/local-briefs/assets and set local_path');
    }
    if (!['business_photo', 'licensed_stock', 'generated_illustration'].includes(photo.provenance)) throw new Error('Invalid media.photos provenance');
    if (photo.provenance === 'licensed_stock' && (!photo.source_url || !photo.license || !photo.credit)) throw new Error('Stock media requires source_url, license and credit');
    const source = path.resolve(localMediaRoot, photo.local_path);
    if (!source.startsWith(localMediaRoot + path.sep) || !existsSync(source)) throw new Error(`Invalid media.photos local_path: ${photo.local_path}`);
  }
  if (scenario !== undefined && (typeof scenario !== 'object' || Array.isArray(scenario))) throw new Error('scenario must be an object');
  if (idempotencyKey !== undefined && (typeof idempotencyKey !== 'string' || !/^[a-zA-Z0-9_-]{8,100}$/.test(idempotencyKey))) throw new Error('Invalid idempotencyKey');
  return { brief, scenario, key: idempotencyKey || digest(brief) };
}

async function runProcess(command, args, options) {
  const { cwd, stdin, stdoutFile, stderrFile, timeoutMs } = options;
  return new Promise(resolve => {
    const out = createWriteStream(stdoutFile, { flags: 'a' });
    const err = createWriteStream(stderrFile, { flags: 'a' });
    let settled = false;
    let child;
    let hardTimer;
    let timeoutReason = null;
    const finish = result => {
      if (settled) return;
      settled = true;
      clearTimeout(timer);
      clearTimeout(hardTimer);
      out.end(); err.end();
      resolve(result);
    };
    let timer;
    try {
      child = spawn(command, args, { cwd, env: cleanEnv(), windowsHide: true, stdio: ['pipe', 'pipe', 'pipe'] });
    } catch (e) { out.end(); err.end(); resolve({ ok: false, reason: `spawn: ${e.message}` }); return; }
    child.stdout.pipe(out, { end: false });
    child.stderr.pipe(err, { end: false });
    child.on('error', e => finish({ ok: false, reason: `spawn: ${e.message}` }));
    child.on('close', (code, signal) => finish({ ok: code === 0 && !timeoutReason, code, signal, reason: timeoutReason || (code === 0 ? null : `exit ${code ?? signal}`) }));
    timer = setTimeout(() => {
      timeoutReason = `timeout after ${timeoutMs}ms`;
      child.kill('SIGTERM');
      hardTimer = setTimeout(() => { child.kill('SIGKILL'); finish({ ok: false, reason: timeoutReason }); }, 5000);
    }, timeoutMs);
    if (stdin) child.stdin.end(stdin); else child.stdin.end();
  });
}

async function processJob(job) {
  const dir = jobPath(job.id);
  const site = path.join(dir, 'site');
  const publicDir = path.join(site, 'public');
  await mkdir(publicDir, { recursive: true });
  const photos = job.brief.media?.photos || [];
  const mediaDir = path.join(publicDir, 'media');
  if (photos.length) await mkdir(mediaDir, { recursive: true });
  const manifest = [];
  const canonicalRoot = photos.length ? await realpath(localMediaRoot) : null;
  for (const [index, photo] of photos.entries()) {
    const source = await realpath(path.resolve(localMediaRoot, photo.local_path));
    if (!source.startsWith(canonicalRoot + path.sep)) throw new Error(`Media file escapes asset root: ${photo.local_path}`);
    const info = await stat(source);
    if (!info.isFile() || info.size > 8_000_000) throw new Error(`Media file missing or above 8 MB: ${photo.local_path}`);
    const output = `photo-${index + 1}${path.extname(source).toLowerCase()}`;
    await copyFile(source, path.join(mediaDir, output));
    manifest.push({ file: `media/${output}`, caption: photo.caption || '', provenance: photo.provenance, credit: photo.credit || null, license: photo.license || null, source_url: photo.source_url || null, source: photo.local_path, width: photo.width || null, height: photo.height || null });
  }
  const mediaImageArgs = manifest.slice(0, 6).flatMap(item => ['--image', path.join(publicDir, item.file)]);
  const publicFonts = path.join(publicDir, 'fonts');
  await mkdir(publicFonts, { recursive: true });
  for (const name of (await readdir(fontKit)).filter(name => /\.(woff2|css|txt)$/.test(name))) {
    await copyFile(path.join(fontKit, name), path.join(publicFonts, name));
  }
  await mkdir(path.join(dir, 'qa'), { recursive: true });

async function drain() {
  if (busy || closing) return;
  const next = [...jobs.values()].find(job => job.state === 'queued' && (job.ownerPid === process.pid || !liveOwner(job.ownerPid)));
  if (!next) return;
  busy = true;
  try { await processJob(next); }
  catch (error) {
    next.state = 'failed'; next.error = `Job setup failed: ${error.message}`; next.completedAt = new Date().toISOString(); next.ownerPid = null;
    await save(next).catch(console.error);
  } finally { busy = false; if (!closing) setImmediate(drain); }
}
async function init() {
  await mkdir(jobsDir, { recursive: true });
  for (const id of await readdir(jobsDir)) {
    if (!/^[0-9a-f-]{36}$/.test(id)) continue;
    try {
      const job = JSON.parse(await readFile(path.join(jobPath(id), 'job.json'), 'utf8'));
      if (job.id !== id) continue;
      if (job.state === 'running' && !liveOwner(job.ownerPid)) { job.state = 'interrupted'; job.error = 'Server stopped during run; submit a new key to retry'; job.ownerPid = null; await save(job); }
      jobs.set(id, job);
    } catch { /* Invalid partial job stays on disk for inspection. */ }
  }
}
async function main() {
  await init();
  const server = createServer(async (req, res) => {
    try {
      const url = new URL(req.url, `http://${host}:${port}`);
      if (req.method === 'GET' && url.pathname === '/health') return respond(res, 200, { service: 'frontstep-local', status: 'ok', pid: process.pid, queue: [...jobs.values()].filter(j => j.state === 'queued').length, running: busy, externalSystems: 'not checked' });
      if (req.method === 'GET' && url.pathname === '/jobs') return respond(res, 200, [...jobs.values()].map(publicJob));
      const match = url.pathname.match(/^\/jobs\/([0-9a-f-]{36})$/);
      if (req.method === 'GET' && match) {
        const job = jobs.get(match[1]); return job ? respond(res, 200, publicJob(job)) : respond(res, 404, { error: 'Job not found' });
      }
      const retry = url.pathname.match(/^\/jobs\/([0-9a-f-]{36})\/retry$/);
      if (req.method === 'POST' && retry) {
        const job = jobs.get(retry[1]);
        if (!job) return respond(res, 404, { error: 'Job not found' });
        if (!['failed', 'interrupted'].includes(job.state)) return respond(res, 409, { error: `Cannot retry ${job.state} job` });
        if ((job.retryCount || 0) >= 2) return respond(res, 409, { error: 'Retry limit reached' });
        job.retryCount = (job.retryCount || 0) + 1; job.state = 'queued'; job.error = null; job.completedAt = null; job.ownerPid = process.pid;
        await save(job); setImmediate(drain); return respond(res, 202, publicJob(job));
      }
      if (req.method === 'POST' && url.pathname === '/jobs') {
        const { brief, scenario, key } = validate(await bodyJson(req));
        const prior = [...jobs.values()].find(j => j.key === key);
        const requestHash = digest({ brief, scenario });
        if (prior) return respond(res, prior.requestHash === requestHash ? 200 : 409, prior.requestHash === requestHash ? publicJob(prior) : { error: 'Idempotency key used for different request' });
        const id = randomUUID();
        const job = { id, key, requestHash, brief, scenario, state: 'queued', createdAt: new Date().toISOString(), startedAt: null, completedAt: null, attempt: 0, retryCount: 0, error: null, ownerPid: process.pid };
        await mkdir(jobPath(id), { recursive: false }); await save(job); jobs.set(id, job);
        setImmediate(drain); return respond(res, 202, publicJob(job));
      }
      const preview = url.pathname.match(/^\/preview\/([0-9a-f-]{36})(\/.*)?$/);
      if (req.method === 'GET' && preview) {
        const job = jobs.get(preview[1]);
        if (!job || job.state !== 'succeeded') return respond(res, 404, { error: 'Preview unavailable' });
        const root = path.join(jobPath(job.id), 'site', 'public');
        const eligibility = await previewEligibility(job, root, publicDigest);
        if (!eligibility.ok) return respond(res, 409, { error: eligibility.reason });
        const relative = decodeURIComponent(preview[2] || '/').replace(/^\/+/, '') || 'index.html';
        const target = path.resolve(root, relative);
        if (!target.startsWith(root + path.sep) && target !== path.join(root, 'index.html')) return respond(res, 403, { error: 'Forbidden' });
        const info = await stat(target).catch(() => null);
        if (!info?.isFile()) return respond(res, 404, { error: 'Not found' });
        const actual = await realpath(target);
        if (!actual.startsWith(root + path.sep)) return respond(res, 403, { error: 'Forbidden' });
        const mime = { '.html': 'text/html', '.css': 'text/css', '.js': 'text/javascript', '.svg': 'image/svg+xml', '.png': 'image/png', '.jpg': 'image/jpeg', '.webp': 'image/webp', '.woff2': 'font/woff2' }[path.extname(target)] || 'application/octet-stream';
        res.writeHead(200, { 'content-type': mime, 'x-content-type-options': 'nosniff' }); res.end(await readFile(target)); return;
      }
      respond(res, 404, { error: 'Not found' });
    } catch (e) { respond(res, e instanceof SyntaxError || /brief|Body|scenario|Invalid|Synthetic|Missing media|Missing business|media\.photos/.test(e.message) ? 400 : 500, { error: e.message }); }
  });
  server.listen(port, host, () => { console.log(`Frontstep local: http://${host}:${port}`); drain(); });
  for (const signal of ['SIGINT', 'SIGTERM']) process.on(signal, () => { closing = true; server.close(() => process.exit(0)); });
}
main().catch(e => { console.error(e); process.exitCode = 1; });
