#!/usr/bin/env node
// Serves completed local examples and a small index. Never deploys or accepts writes.
import { createServer } from 'node:http';
import { readFile, readdir, realpath, stat } from 'node:fs/promises';
import { createHash } from 'node:crypto';
import path from 'node:path';
import { fileURLToPath } from 'node:url';

const here = path.dirname(fileURLToPath(import.meta.url));
const examplesRoot = path.join(here, 'examples');
const jobsRoot = path.join(here, 'jobs');
const catalog = JSON.parse(await readFile(path.join(examplesRoot, 'catalog.json'), 'utf8'));
const catalogBriefs = new Map(catalog.map(entry => [entry.brief, entry]));
const port = Number(process.env.FRONTSTEP_EXAMPLES_PORT || 4191);
const mime = { '.html': 'text/html; charset=utf-8', '.css': 'text/css; charset=utf-8', '.js': 'text/javascript; charset=utf-8', '.json': 'application/json; charset=utf-8', '.png': 'image/png', '.jpg': 'image/jpeg', '.jpeg': 'image/jpeg', '.webp': 'image/webp', '.svg': 'image/svg+xml', '.woff2': 'font/woff2', '.ico': 'image/x-icon', '.avif': 'image/avif', '.gif': 'image/gif' };
const escape = value => String(value).replace(/[&<>"']/g, char => ({ '&': '&amp;', '<': '&lt;', '>': '&gt;', '"': '&quot;', "'": '&#39;' })[char]);
const criteria = ['artDirection', 'typography', 'composition', 'imagery', 'motion', 'interaction', 'responsive'];
const uuid = /^[0-9a-f]{8}-[0-9a-f]{4}-[0-9a-f]{4}-[0-9a-f]{4}-[0-9a-f]{12}$/;

async function fileInside(root, requested) {
  const canonicalRoot = await realpath(root).catch(() => null);
  if (!canonicalRoot) return null;
  const target = path.resolve(root, requested);
  const canonical = await realpath(target).catch(() => null);
  if (!canonical || !canonical.startsWith(canonicalRoot + path.sep)) return null;
  const info = await stat(canonical).catch(() => null);
  return info?.isFile() ? canonical : null;
}

async function publicDigest(root) {
  const hash = createHash('sha256');
  async function walk(folder, prefix = '') {
    for (const entry of (await readdir(folder, { withFileTypes: true })).sort((a, b) => a.name.localeCompare(b.name))) {
      const relative = prefix + entry.name;
      if (entry.isDirectory()) await walk(path.join(folder, entry.name), relative + '/');
      else if (entry.isFile()) {
        hash.update(relative + '\0');
        hash.update(await readFile(path.join(folder, entry.name)));
      }
    }
  }
  await walk(root);
  return hash.digest('hex');
}

async function approvedAutomatic() {
  const selected = new Map();
  const canonicalJobs = await realpath(jobsRoot).catch(() => null);
  if (!canonicalJobs) return selected;
  for (const entry of await readdir(jobsRoot, { withFileTypes: true })) {
    if (!entry.isDirectory() || !uuid.test(entry.name)) continue;
    const dir = path.join(jobsRoot, entry.name);
    if (await realpath(dir).catch(() => null) !== path.join(canonicalJobs, entry.name)) continue;
    const jobFile = await fileInside(dir, 'job.json');
    if (!jobFile) continue;
    const job = await readFile(jobFile, 'utf8').then(JSON.parse).catch(() => null);
    if (!job || job.id !== entry.name || job.state !== 'succeeded' || job.qa?.passed !== true ||
      job.visual?.passed !== true || !(job.visual.score > 9) ||
      !criteria.every(key => Number.isFinite(job.visual.criteria?.[key]) && job.visual.criteria[key] >= 8.5) ||
      job.motion?.signatureEvidence?.hasMeaningfulChange !== true ||
      !/^[0-9a-f]{64}$/.test(job.publicSha256 || '') ||
      typeof job.pipelineVersion !== 'string' || !job.pipelineVersion ||
      job.brief?.synthetic_fixture !== true) continue;
    const catalogEntry = catalogBriefs.get(job.brief.slug);
    if (!catalogEntry) continue;
    const publicRoot = path.join(dir, 'site', 'public');
    const canonicalDir = await realpath(dir);
    const canonicalPublic = await realpath(publicRoot).catch(() => null);
    const canonicalQa = await realpath(path.join(dir, 'qa')).catch(() => null);
    if (!canonicalPublic?.startsWith(canonicalDir + path.sep) || !canonicalQa?.startsWith(canonicalDir + path.sep)) continue;
    const index = await fileInside(publicRoot, 'index.html');
    const shot = await fileInside(path.join(dir, 'qa'), 'desktop.png');
    if (!index || !shot) continue;
    if (await publicDigest(publicRoot).catch(() => null) !== job.publicSha256) continue;
    const timestamp = Date.parse(job.completedAt || job.createdAt || '') || 0;
    if (!selected.has(catalogEntry.brief) || timestamp > selected.get(catalogEntry.brief).timestamp) {
      selected.set(catalogEntry.brief, { id: job.id, entry: catalogEntry, timestamp, dir });
    }
  }
  return selected;
}

      res.writeHead(200, { 'content-type': mime['.html'], 'x-content-type-options': 'nosniff', 'cache-control': 'no-store' });

      if (!job) { res.writeHead(404).end(); return; }
      const requested = automaticSite[2] || 'index.html';
      const target = await fileInside(path.join(job.dir, 'site', 'public'), requested);
      const extension = path.extname(target || '').toLowerCase();
      const allowed = ['.html', '.css', '.js', '.png', '.jpg', '.jpeg', '.webp', '.svg', '.woff2', '.json', '.ico', '.avif', '.gif'];
      if (!target || !allowed.includes(extension)) { res.writeHead(404).end(); return; }
      res.writeHead(200, { 'content-type': mime[extension] || 'application/octet-stream', 'x-content-type-options': 'nosniff', 'cache-control': 'no-store' });
      res.end(await readFile(target)); return;
    }
    const thumb = pathname.match(/^\/thumb\/([a-z0-9-]+)\.png$/);
    if (thumb) {
