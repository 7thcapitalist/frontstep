#!/usr/bin/env node
// PostToolUse hook: registra cada chamada da ferramenta Skill em <build>/skills.jsonl.
// Regra de ouro: NUNCA derrubar um build. Qualquer erro aqui é engolido e sai 0.
import fs from "node:fs";
try {
  const out = process.argv[2] || process.env.FRONTSTEP_SKILLS_LOG;
  if (out) {
    let raw = "";
    process.stdin.setEncoding("utf8");
    for await (const c of process.stdin) { raw += c; if (raw.length > 2e6) break; }
    let p = {}; try { p = JSON.parse(raw); } catch {}
    const i = p.tool_input || {};
    const skill = i.skill || i.name || i.skill_name || null;
    fs.appendFileSync(out, JSON.stringify({
      ts: new Date().toISOString(),
      skill,
      args: typeof i.args === "string" ? i.args.slice(0, 300) : undefined,
      tool_use_id: p.tool_use_id || undefined,
      session_id: p.session_id || undefined
    }) + "\n");
  }
} catch {}
process.exit(0);
